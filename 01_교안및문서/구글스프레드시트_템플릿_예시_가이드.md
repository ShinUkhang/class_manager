## 🔌 [1분 완성] 구글 스프레드시트 실시간 연동 스크립트 (Google Apps Script)

구글 스프레드시트 메뉴에서 **[확장 프로그램] ➜ [Apps Script]**를 클릭하고, 기존 코드를 지운 뒤 아래 코드를 그대로 붙여넣고 **[배포] ➜ [새 배포] ➜ [웹 앱(모든 사용자)]**으로 배포하시면 됩니다.

```javascript
// 웹 대시보드에서 전송한 출결 및 조퇴 데이터를 구글 시트에 실시간 기록하는 스크립트
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 1. 출결 기록 저장
    if (data.type === "attendance") {
      var sheet = ss.getSheetByName("출결기록") || ss.insertSheet("출결기록");
      if (sheet.getLastRow() === 0) {
        sheet.appendRow(["기록일시", "날짜", "번호", "이름", "출결상태", "입력구분", "비고"]);
      }
      sheet.appendRow([
        new Date().toLocaleString("ko-KR", { timeZone: "Asia/Seoul" }),
        data.date,
        data.studentNo,
        data.studentName,
        data.status,
        data.author || "담임인증",
        data.note || ""
      ]);
      return ContentService.createTextOutput(JSON.stringify({ result: "success", message: "출결 저장 완료" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // 2. 조퇴허가증 발급 기록 저장
    if (data.type === "early_leave") {
      var sheet = ss.getSheetByName("조퇴허가증기록") || ss.insertSheet("조퇴허가증기록");
      if (sheet.getLastRow() === 0) {
        sheet.appendRow(["발급일시", "발급번호", "학번", "이름", "조퇴시간", "조퇴사유", "담임서명여부", "보호자인계여부"]);
      }
      sheet.appendRow([
        new Date().toLocaleString("ko-KR", { timeZone: "Asia/Seoul" }),
        data.issueNo,
        data.studentId,
        data.studentName,
        data.leaveTime,
        data.reason,
        "서명완료",
        data.guardianContact || "확인완료"
      ]);
      return ContentService.createTextOutput(JSON.stringify({ result: "success", message: "조퇴증 대장 기록 완료" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({ result: "ignored" }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", error: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// 구글 시트 데이터를 웹 대시보드로 읽어오는 함수 (시간표, 급식 등 조회용)
function doGet(e) {
  var sheetName = e.parameter.sheet || "시간표";
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName);
  if (!sheet) {
    return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  }
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var rows = [];
  for (var i = 1; i < data.length; i++) {
    var row = {};
    for (var j = 0; j < headers.length; j++) {
      row[headers[j]] = data[i][j];
    }
    rows.push(row);
  }
  return ContentService.createTextOutput(JSON.stringify(rows)).setMimeType(ContentService.MimeType.JSON);
}
```
